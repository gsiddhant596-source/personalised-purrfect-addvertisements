import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).parent / "adsense_ai.db"


def get_connection():
    connection = sqlite3.connect(DB_PATH)
    connection.row_factory = sqlite3.Row
    return connection


def create_tables():
    connection = get_connection()

    # User search tracking
    connection.execute("""
        CREATE TABLE IF NOT EXISTS search_history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            query TEXT NOT NULL,
            searched_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)

    # Company & Product Advertising Catalog
    connection.execute("""
        CREATE TABLE IF NOT EXISTS ad_campaigns (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            company_name TEXT NOT NULL,
            product_name TEXT NOT NULL,
            category TEXT NOT NULL,
            description TEXT NOT NULL,
            target_topics TEXT NOT NULL,
            price TEXT,
            image_url TEXT,
            destination_url TEXT NOT NULL,
            active INTEGER DEFAULT 1,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)

    connection.commit()
    connection.close()


def save_search(query: str):
    with get_connection() as connection:
        connection.execute(
            "INSERT INTO search_history (query) VALUES (?)",
            (query,)
        )


def get_search_history():
    with get_connection() as connection:
        rows = connection.execute("""
            SELECT id, query, searched_at
            FROM search_history
            ORDER BY id DESC
        """).fetchall()

        return [dict(row) for row in rows]


def add_product_campaign(
    company_name: str,
    product_name: str,
    category: str,
    description: str,
    target_topics: str,
    price: str = "Free",
    image_url: str = None,
    destination_url: str = "https://example.com"
):
    """Adds a complete company product listing."""
    with get_connection() as connection:
        cursor = connection.execute("""
            INSERT INTO ad_campaigns (
                company_name, product_name, category,
                description, target_topics, price,
                image_url, destination_url
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            company_name, product_name, category,
            description, target_topics, price,
            image_url, destination_url
        ))
        connection.commit()
        return cursor.lastrowid


# Backward compatibility wrapper for old seed scripts
def add_ad_campaign(
    name,
    description,
    target_topics,
    image_url=None,
    destination_url=None
):
    add_product_campaign(
        company_name=name,
        product_name=name,
        category="General",
        description=description,
        target_topics=target_topics,
        price="",
        image_url=image_url,
        destination_url=destination_url or "https://example.com"
    )


def get_active_ads():
    with get_connection() as connection:
        rows = connection.execute("""
            SELECT * FROM ad_campaigns
            WHERE active = 1
            ORDER BY id DESC
        """).fetchall()

        ads = []
        for row in rows:
            d = dict(row)
            # Ensure extension compatibility expecting ad["name"]
            if "name" not in d or not d["name"]:
                d["name"] = f"{d.get('product_name', '')} by {d.get('company_name', '')}"
            ads.append(d)
        return ads

@app.get("/search")
def search(topic: str = Query(..., min_length=1, max_length=200)):
    topic = topic.strip()
    if not topic:
        return {"error": "Please enter a search topic."}

    save_search(topic)
    url = "https://en.wikipedia.org/api/rest_v1/page/summary/" + quote(topic.replace(" ", "_"))

    try:
        response = requests.get(url, timeout=10, headers={"User-Agent": "AdSenseAI/1.0"})
        if response.status_code == 200:
            data = response.json()
            return {
                "topic": topic,
                "title": data.get("title", topic),
                "summary": data.get("extract", "No summary available."),
                "article_url": data.get("content_urls", {}).get("desktop", {}).get("page"),
            }
        return {
            "topic": topic,
            "title": topic,
            "summary": "No Wikipedia summary found. Try another search term.",
            "article_url": None,
        }
    except requests.RequestException:
        return {
            "topic": topic,
            "title": topic,
            "summary": "Information service temporarily unavailable.",
            "article_url": None,
        }

class ProductCreate(BaseModel):

    product_name: str

    target_topics: str
    price: str = "Free"
    destination_url: str = "https://example.com"
    custom_image_url: str | None = None


@app.post("/api/company/add-product")
def create_product(data: ProductCreate):
    # Use custom image if provided; otherwise generate locally using SD-Turbo on RTX 4050
    if data.custom_image_url and data.custom_image_url.strip():
        img_url = data.custom_image_url.strip()
    else:
        img_url = generate_local_sd_banner(
            data.product_name,
            data.category,
            data.description
        )

    product_id = add_product_campaign(
        company_name=data.company_name,
        product_name=data.product_name,
        category=data.category,
        description=data.description,
        target_topics=data.target_topics,
        price=data.price,
        image_url=img_url,
        destination_url=data.destination_url
    )

    return {
        "status": "success",
        "product_id": product_id,
        "image_url": img_url
    }

import time
from sentence_transformers import SentenceTransformer, util
from database import get_active_ads

# Initialize semantic embedding model
model = SentenceTransformer("all-MiniLM-L6-v2")

cached_ads = []
cached_embeddings = None
last_cache_refresh = 0
CACHE_TTL_SECONDS = 30


def refresh_ad_cache():
    global cached_ads, cached_embeddings, last_cache_refresh

    cached_ads = get_active_ads()

    if not cached_ads:
        cached_embeddings = None
        last_cache_refresh = time.time()
        return

    ad_texts = []
    for ad in cached_ads:
        topics = ad.get("target_topics", "")
        if isinstance(topics, list):
            topics = ", ".join(topics)

        name = ad.get("product_name") or ad.get("name", "")
        company = ad.get("company_name", "")
        category = ad.get("category", "")
        desc = ad.get("description", "")

        ad_texts.append(
            f"Product: {name} by {company}. Category: {category}. "
            f"Description: {desc}. Target topics and interests: {topics}."
        )

    cached_embeddings = model.encode(
        ad_texts,
        convert_to_tensor=True,
        normalize_embeddings=True
    )
    last_cache_refresh = time.time()


def recommend_ads(page_content: str, limit: int = 2):
    global cached_ads, cached_embeddings, last_cache_refresh

    page_content = " ".join(page_content.split())
    if not page_content:
        return []

    # Refresh periodically so database changes are picked up
    if (
        cached_embeddings is None
        or time.time() - last_cache_refresh >= CACHE_TTL_SECONDS
    ):
        refresh_ad_cache()

    if not cached_ads or cached_embeddings is None:
        return []

    # Encode active page text (first 4000 characters)
    page_embedding = model.encode(
        page_content[:4000],
        convert_to_tensor=True,
        normalize_embeddings=True
    )

    scores = util.cos_sim(page_embedding, cached_embeddings)[0]

    ranked_ads = []
    for index, ad in enumerate(cached_ads):
        score = float(scores[index].item())
        ranked_ads.append({
            "id": ad["id"],
            "name": ad.get("product_name") or ad.get("name", "Sponsored Recommendation"),
            "description": ad.get("description", ""),
            "price": ad.get("price", ""),
            "image_url": ad.get("image_url"),
            "destination_url": ad.get("destination_url"),
            "similarity": round(score, 4),
        })

    ranked_ads.sort(key=lambda ad: ad["similarity"], reverse=True)

    # Reliable threshold for web pages
    MIN_SIMILARITY = 0.15
    matches = [ad for ad in ranked_ads if ad["similarity"] >= MIN_SIMILARITY]

    # Graceful demo fallback: If browsing a mixed/generic page (e.g. YouTube homepage),
    # return the highest-scoring ad instead of leaving the UI blank
    if not matches and ranked_ads:
        return [ranked_ads[0]]

    return matches[:max(1, min(limit, 2))]

import torch
from diffusers import AutoPipelineForText2Image
from pathlib import Path
import uuid

# Setup static directory to save generated ads
STATIC_DIR = Path(__file__).parent / "static" / "ads"
STATIC_DIR.mkdir(parents=True, exist_ok=True)

# Select GPU device
device = "cuda" if torch.cuda.is_available() else "cpu"
torch_dtype = torch.float16 if device == "cuda" else torch.float32

print(f"[Stable Diffusion] Loading SD-Turbo on {device} (VRAM-optimized for 6GB)...")

# Fast 1-step distilled Stable Diffusion model
pipe = AutoPipelineForText2Image.from_pretrained(
    "stabilityai/sd-turbo",
    torch_dtype=torch_dtype,
    variant="fp16" if device == "cuda" else None
)
pipe.to(device)

# Enable memory optimizations
if device == "cuda":
    pipe.enable_attention_slicing()

print("[Stable Diffusion] Model ready for generation.")


def generate_local_sd_banner(product_name: str, category: str, description: str) -> str:
    """
    Generates a commercial ad graphic locally on RTX 4050,
    saves it to disk, and returns the accessible static URL.
    """
    prompt = (
        f"commercial ad banner for {product_name}, {category}, {description}, "
        f"isolated studio lighting, award-winning commercial product photography, 8k, sharp focus"
    )

    # 1-step inference on SD-Turbo runs in ~0.4s on RTX 4050
    image = pipe(
        prompt=prompt,
        num_inference_steps=1,
        guidance_scale=0.0,
        width=512,
        height=320
    ).images[0]

    filename = f"ad_{uuid.uuid4().hex[:8]}.png"
    filepath = STATIC_DIR / filename
    image.save(filepath)

    # Returns the URL hosted by FastAPI
    return f"http://127.0.0.1:8000/static/ads/{filename}"

const API_BASE = "http://127.0.0.1:8000";

const consentButton = document.getElementById("consentButton");
const analyseButton = document.getElementById("analyseButton");
const stopButton = document.getElementById("stopButton");
const statusBox = document.getElementById("status");
const adsBox = document.getElementById("ads");

function setStatus(message) {
  statusBox.textContent = message;
}

// Fallback graphic for missing images
function createAdPlaceholder(adName = "AdSense AI") {
  const placeholder = document.createElement("div");
  placeholder.className = "ad-image-placeholder";

  const icon = document.createElement("div");
  icon.className = "placeholder-icon";
  icon.textContent = "✦";

  const brand = document.createElement("div");
  brand.className = "placeholder-brand";
  brand.textContent = adName;

  placeholder.append(icon, brand);
  return placeholder;
}

function showAds(ads) {
  adsBox.replaceChildren();

  if (!ads.length) {
    setStatus("No matching ads found for this page.");
    return;
  }

  for (const ad of ads) {
    const card = document.createElement("article");
    card.className = "ad-card";

    // Ad banner image
    const imageArea = document.createElement("div");
    if (ad.image_url) {
      const image = document.createElement("img");
      image.alt = ad.name || "Advertisement";
      image.referrerPolicy = "no-referrer";
      image.loading = "eager";
      image.src = ad.image_url;
      image.onerror = () => {
        imageArea.replaceChildren(createAdPlaceholder(ad.name));
      };
      imageArea.appendChild(image);
    } else {
      imageArea.appendChild(createAdPlaceholder(ad.name));
    }
    card.appendChild(imageArea);

    // Ad text content
    const content = document.createElement("div");
    content.className = "ad-content";

    const matchPercent = Math.round((ad.similarity || 0.75) * 100);
    const label = document.createElement("div");
    label.className = "sponsored";
    label.textContent = `✦ Semantic Match (${matchPercent}%)`;

    const title = document.createElement("h2");
    title.textContent = ad.name || "Sponsored Recommendation";

    const description = document.createElement("p");
    description.textContent = ad.description || "";

    content.append(label, title, description);

    // Destination link
    if (ad.destination_url) {
      const link = document.createElement("a");
      link.href = ad.destination_url;
      link.target = "_blank";
      link.rel = "noopener noreferrer";
      link.textContent = "Learn more ↗";
      content.appendChild(link);
    }

    card.appendChild(content);
    adsBox.appendChild(card);
  }

  setStatus(`${ads.length} relevant ad${ads.length === 1 ? "" : "s"} matched.`);
}

async function readSession() {
  return chrome.storage.session.get(["consent"]);
}

async function refreshControls() {
  const data = await readSession();
  const allowed = data.consent === true;

  consentButton.hidden = allowed;
  analyseButton.disabled = !allowed;
  stopButton.hidden = !allowed;
}

// Enable consent
consentButton.addEventListener("click", async () => {
  await chrome.storage.session.set({ consent: true });
  adsBox.replaceChildren();
  await refreshControls();
  setStatus("Consent active. Click 'Find relevant ads' to match page content.");
});

// Disable consent
stopButton.addEventListener("click", async () => {
  await chrome.storage.session.set({ consent: false });
  adsBox.replaceChildren();
  await refreshControls();
  setStatus("Consent turned off. No ads will be displayed.");
});

// Read page and fetch recommendation
analyseButton.addEventListener("click", async () => {
  analyseButton.disabled = true;
  adsBox.replaceChildren();
  setStatus("Extracting webpage text...");

  try {
    const session = await readSession();
    if (session.consent !== true) {
      setStatus("Please grant session permission first.");
      return;
    }

    const [tab] = await chrome.tabs.query({
      active: true,
      currentWindow: true
    });

    if (!tab || !tab.id || !/^https?:/.test(tab.url || "")) {
      setStatus("Open a standard website first (browser internal tabs cannot be analyzed).");
      return;
    }

    const results = await chrome.scripting.executeScript({
      target: { tabId: tab.id },
      func: () => ({
        title: document.title || "",
        text: (document.body?.innerText || "").slice(0, 8000)
      })
    });

    const page = results?.[0]?.result;
    if (!page || !page.text.trim()) {
      setStatus("No readable text found on this page.");
      return;
    }

    const pageContent = `${page.title}\n${page.text}`.slice(0, 6000);
    setStatus("Matching with AI vector database...");

    const endpoint = new URL(`${API_BASE}/recommend`);
    endpoint.searchParams.set("content", pageContent);

    const response = await fetch(endpoint.toString());
    if (!response.ok) {
      throw new Error(`Server returned HTTP ${response.status}`);
    }

    const data = await response.json();
    const recommendations = Array.isArray(data) ? data : data.recommendations;

    if (!Array.isArray(recommendations) || recommendations.length === 0) {
      setStatus("No matching ads found in inventory.");
      return;
    }

    showAds(recommendations);
  } catch (error) {
    console.error("AdSense AI error:", error);
    if (error.message?.includes("Failed to fetch")) {
      setStatus("Cannot reach FastAPI backend. Verify 'python main.py' is running on port 8000.");
    } else {
      setStatus(`Error: ${error.message || "Failed to analyze page."}`);
    }
  } finally {
    analyseButton.disabled = false;
  }
});

// Initialize extension popup
refreshControls().then(() => {
  readSession().then(data => {
    if (data.consent === true) {
      setStatus("Ready. Click 'Find relevant ads' to match page content.");
    } else {
      setStatus("Consent is required before analysis.");
    }
  });
});

        const API = "http://127.0.0.1:8000";

        async function searchTopic(topic) {
            topic = topic.trim();

            if (!topic) return;

            input.value = topic;
            button.disabled = true;
            status.textContent = "Searching...";
            resultPanel.hidden = true;

            try {
                const response = await fetch(
                    `${API}/search?topic=${encodeURIComponent(topic)}`
                );

                if (!response.ok) {
                    throw new Error("The search request failed.");
                }

                const data = await response.json();

                document.getElementById("resultTitle").textContent =
                    data.title || topic;

                document.getElementById("summary").textContent =
                    data.summary || "No summary is available.";

                const articleLink = document.getElementById("articleLink");

                if (data.article_url) {
                    articleLink.href = data.article_url;
                    articleLink.hidden = false;
                } else {
                    articleLink.hidden = true;
                    articleLink.removeAttribute("href");
                }

                resultPanel.hidden = false;
                status.textContent = "";
                await loadHistory();

            } catch (error) {
                status.textContent =
                    "Couldn't complete the search. Check that the backend is running.";
            } finally {
                button.disabled = false;
            }
        }

        async function loadHistory() {
            const historyList = document.getElementById("historyList");

            try {
                const response = await fetch(`${API}/history`);

                if (!response.ok) {
                    throw new Error("History request failed.");
                }

                const data = await response.json();
                historyList.replaceChildren();

                if (!data.history || data.history.length === 0) {
                    historyList.textContent = "No searches yet.";
                    return;

                data.history.forEach(item => {

                    chip.type = "button";
                    chip.className = "history-item";
                    chip.textContent = item.query;

                    chip.addEventListener("click", () => {
                        searchTopic(item.query);
                    });

                    historyList.appendChild(chip);
                });

            } catch {
                historyList.textContent =
                    "History is unavailable. Check the backend.";
            }
        }

            event.preventDefault();
            searchTopic(input.value);

                searchTopic(button.dataset.query);
            });
        });
